#include <tunables/global>

profile donetick flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  capability net_bind_service,
  network inet stream,
  network inet6 stream,

  signal (send,receive) peer=*_donetick,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /dev/std{out,err} rw,
  /proc/*/fd/* rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Addon persistent data (sqlite db, options.json)
  /data/** rw,

  # donetick backend + entrypoint script
  /app/** rwix,

  # nginx: config is read-only at runtime (baked in at build time), everything
  # else under /var/lib/nginx and /run/nginx is its runtime scratch space
  /etc/nginx/** r,
  /var/lib/nginx/** rwk,
}
